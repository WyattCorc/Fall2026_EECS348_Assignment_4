#include <iostream>
#include <fstream>
#include <vector>
#include <string>
#include <set>

class Grid {
private:
    char board[9][9];

public:
    Grid() {
        for (int r = 0; r < 9; ++r) {
            for (int c = 0; c < 9; ++c) {
                board[r][c] = '_';
            }
        }
    }

    // Load board from a file
    bool loadFromFile(const std::string& filename) {
        std::ifstream file(filename);
        if (!file.is_open()) {
            return false;
        }
        for (int r = 0; r < 9; ++r) {
            for (int c = 0; c < 9; ++c) {
                if (!(file >> board[r][c])) {
                    return false;
                }
            }
        }
        return true;
    }

    // Print the grid in a clean 9x9 format with block separators
    void print() const {
        for (int r = 0; r < 9; ++r) {
            if (r > 0 && r % 3 == 0) {
                std::cout << "------+-------+------\n";
            }
            for (int c = 0; c < 9; ++c) {
                if (c > 0 && c % 3 == 0) {
                    std::cout << "| ";
                }
                std::cout << board[r][c] << ' ';
            }
            std::cout << '\n';
        }
        std::cout << '\n';
    }

    bool isEmptyCell(int r, int c) const {
        return board[r][c] == '_';
    }

    char getCell(int r, int c) const {
        return board[r][c];
    }

    void setCell(int r, int c, char val) {
        board[r][c] = val;
    }
};

class Solver {
private:
    // Find the next empty cell (returns true if found, false if board is full)
    bool findEmptyCell(const Grid& grid, int& row, int& col) const {
        for (row = 0; row < 9; ++row) {
            for (col = 0; col < 9; ++col) {
                if (grid.isEmptyCell(row, col)) {
                    return true;
                }
            }
        }
        return false;
    }

    // Compute valid candidate digits (1-9) for a specific cell
    std::vector<char> getCandidates(const Grid& grid, int row, int col) const {
        std::set<char> used;

        // Check row and column
        for (int i = 0; i < 9; ++i) {
            used.insert(grid.getCell(row, i));
            used.insert(grid.getCell(i, col));
        }

        // Check 3x3 subgrid block
        int startRow = (row / 3) * 3;
        int startCol = (col / 3) * 3;
        for (int r = 0; r < 3; ++r) {
            for (int c = 0; c < 3; ++c) {
                used.insert(grid.getCell(startRow + r, startCol + c));
            }
        }

        // Collect unused digits ('1' through '9')
        std::vector<char> candidates;
        for (char digit = '1'; digit <= '9'; ++digit) {
            if (used.find(digit) == used.end()) {
                candidates.push_back(digit);
            }
        }
        return candidates;
    }

    // Recursive backtracking helper to find and print all solutions
    void solveRecursive(Grid& grid, int& solutionCount) {
        int row, col;
        // Step 1: Find an empty cell
        if (!findEmptyCell(grid, row, col)) {
            // Base case: All cells filled, solution found!
            solutionCount++;
            std::cout << "Solution " << solutionCount << ":\n";
            grid.print();
            return;
        }

        // Step 2: Compute candidates for this empty cell
        std::vector<char> candidates = getCandidates(grid, row, col);

        // Step 3: If no candidates, backtrack automatically (returns to previous frame)
        if (candidates.empty()) {
            return;
        }

        // Step 4 & 5: Try each candidate in turn and recurse
        for (char candidate : candidates) {
            grid.setCell(row, col, candidate);
            solveRecursive(grid, solutionCount);
            grid.setCell(row, col, '_'); // Backtrack
        }
    }

public:
    // Solve a puzzle and print all solutions
    void solve(Grid& grid, const std::string& filename) {
        std::cout << "========================================\n";
        std::cout << "Processing file: " << filename << "\n";
        std::cout << "========================================\n";
        std::cout << "Original Puzzle:\n";
        grid.print();

        int solutionCount = 0;
        solveRecursive(grid, solutionCount);

        if (solutionCount == 0) {
            std::cout << "No solution found for " << filename << ".\n\n";
        } else {
            std::cout << "Total solutions found: " << solutionCount << "\n\n";
        }
    }
};

int main() {
    Solver solver;
    std::string files[] = {"puzzle1.txt", "puzzle2.txt", "puzzle3.txt", "puzzle4.txt", "puzzle5.txt"};

    for (const std::string& filename : files) {
        Grid grid;
        if (grid.loadFromFile(filename)) {
            solver.solve(grid, filename);
        } else {
            std::cout << "Could not open or read file: " << filename << "\n\n";
        }
    }

    return 0;
}
